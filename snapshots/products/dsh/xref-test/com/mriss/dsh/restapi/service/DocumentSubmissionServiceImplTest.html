109     /**
110      * useCache=false and the file is already stored and processed: a file is stored once, so
111      * the stored document's token is returned and that document is queued again from the start.
112      */
113     @Test
114     public void testGetTokenFromDocumentNoCachingWhenStored_shouldRequeueTheStoredDocument() throws Exception {
115         Document stored = storedDocument();
116         processed(stored);
117         when(docHandlingService.getDocumentByHash(stored.getFileHash())).thenReturn(stored);
118 
119         String token = tokenFor("target/test-classes/pdf/bbc-news-1.pdf",
120                 "Russia-Trump: FBI chief Wray defends agency", false);
121         service.storeDocumentAndQueueForProcessing();
122 
123         assertEquals(stored.getToken(), token);
124         verify(messageHandler).setDocument(stored);
125         verify(docHandlingService).storeDocument(stored);
126         verify(docQueueService).enqueueDocumentId(eq("stored-id"), any());
127         assertEquals(DocumentStatus.QUEUED_FOR_INDEXING, stored.getDocumentStatus());
128     }

171     /**
172      * Happy path: document is stored and the RabbitMQ queue mock immediately
173      * simulates an ACK so that the status transitions to
174      * {@link DocumentStatus#QUEUED_FOR_INDEXING_SUCCESS}.
175      */
176     @Test
177     public void testStoreDocumentAndQueueForProcessingSuccess() throws Exception {
178         // Stub storeDocument to return an id so document.getId() works
179         when(docHandlingService.storeDocument(any(Document.class))).thenReturn("stored-id-001");
180 
181         // Simulate RabbitMQ ACK: call the real message handler with a "sent ok" payload
182         // Note: document.getId() is null because DocumentDao.storeDocument()'s return value
183         // is not used to set the document id field, so we must use nullable() matcher.
184         doAnswer(inv -> {
185             DocumentEnqueueMessageHandler handler = inv.getArgument(1);
186             handler.handleMessage(new GenericMessage<>("stored-id-001 sent ok"));
187             return null;
188         }).when(docQueueService).enqueueDocumentId(org.mockito.ArgumentMatchers.nullable(String.class), any());
189 
190         tokenFor("target/test-classes/pdf/bbc-news-1.pdf",
191                 "Russia-Trump: FBI chief Wray defends agency", false);
192         service.storeDocumentAndQueueForProcessing();
193 
194         verify(docHandlingService).storeDocument(any(Document.class));
195         verify(docQueueService).enqueueDocumentId(org.mockito.ArgumentMatchers.nullable(String.class), any());
196     }

253     private String tokenFor(String path, String title, boolean useCache) throws Exception {
254         try (InputStream is = new FileInputStream(new File(path))) {
255             return service.getTokenFromDocument(is, title, useCache);
256         }
257     }

259     private static Document documentFrom(String path, String title) throws Exception {
260         try (InputStream is = new FileInputStream(new File(path))) {
261             return new Document(is, title);
262         }
263     }

265     /** The bbc-news-1 document as the store holds it after one round: an id, queued successfully. */
266     private static Document storedDocument() throws Exception {
267         Document stored = documentFrom("target/test-classes/pdf/bbc-news-1.pdf",
268                 "Russia-Trump: FBI chief Wray defends agency");
269         stored.setId("stored-id");
270         stored.transitionStatus(TransitionType.NEUTRAL);
271         stored.transitionStatus(TransitionType.SUCCESS);
272         return stored;
273     }

275     private static Document processed(Document document) {
276         document.addKeyword(new Keyword("wray", 5.0));
277         document.addSentence(new Sentence("FBI chief Wray defends agency", 4.0, 1));
278         return document;
279     }
