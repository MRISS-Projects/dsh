17      @Test
18      public void getStatusDescription_forEveryStatus_shouldEqualItsName() {
19          for (DocumentStatus status : DocumentStatus.values()) {
20              assertEquals(status.name(), status.getStatusDescription());
21          }
22      }

24      @Test
25      public void getStatusMessage_forEveryStatus_shouldBeItsOwn() {
26          Map<String, DocumentStatus> owners = new HashMap<>();
27          for (DocumentStatus status : DocumentStatus.values()) {
28              String message = status.getStatusMessage();
29              assertNotNull(status + " has no message", message);
30              assertFalse(status + " has a blank message", message.trim().isEmpty());
31              DocumentStatus owner = owners.putIfAbsent(message, status);
32              assertNull(status + " shares its message with " + owner, owner);
33          }
34      }

36      @Test
37      public void getStatusMessage_ofNotIndexedError_shouldSayThatIndexingFailed() {

39      }

41      @Test
42      public void testTransition() {
43  
44          DocumentStatus status = DocumentStatus.CREATED;
45          status = status.transition(TransitionType.NEUTRAL);
46          assertEquals(DocumentStatus.QUEUED_FOR_INDEXING, status);
47  
48          // NEUTRAL on QUEUED_FOR_INDEXING returns self
49          DocumentStatus neutral = status.transition(TransitionType.NEUTRAL);
50          assertSame(status, neutral);
51  
52          status = status.transition(TransitionType.ERROR);
53          assertEquals(DocumentStatus.QUEUED_FOR_INDEXING_ERROR, status);
54          DocumentStatus status1 = status.transition(TransitionType.NEUTRAL);
55          assertTrue(status == status1);
56  
57          status = DocumentStatus.QUEUED_FOR_INDEXING;
58          status = status.transition(TransitionType.SUCCESS);
59          assertEquals(DocumentStatus.QUEUED_FOR_INDEXING_SUCCESS, status);
60  
61          status = status.transition(TransitionType.NEUTRAL);
62          assertEquals(DocumentStatus.DEQUEUED_FOR_INDEXING, status);
63  
64          status = status.transition(TransitionType.NEUTRAL);
65          assertEquals(DocumentStatus.INDEXING, status);
66  
67          // NEUTRAL on INDEXING returns self
68          assertSame(DocumentStatus.INDEXING, DocumentStatus.INDEXING.transition(TransitionType.NEUTRAL));
69  
70          status = status.transition(TransitionType.ERROR);
71          assertEquals(DocumentStatus.NOT_INDEXED_ERROR, status);
72          status1 = status.transition(TransitionType.NEUTRAL);
73          assertTrue(status == status1);
74  
75          status = DocumentStatus.INDEXING;
76          status = status.transition(TransitionType.SUCCESS);
77          assertEquals(DocumentStatus.INDEXED_SUCCESS, status);
78  
79          status = status.transition(TransitionType.NEUTRAL);
80          assertEquals(DocumentStatus.QUEUED_FOR_PROCESSING, status);
81  
82          // NEUTRAL on QUEUED_FOR_PROCESSING returns self
83          assertSame(DocumentStatus.QUEUED_FOR_PROCESSING,
84                  DocumentStatus.QUEUED_FOR_PROCESSING.transition(TransitionType.NEUTRAL));
85  
86          status = status.transition(TransitionType.ERROR);
87          assertEquals(DocumentStatus.QUEUED_FOR_PROCESSING_ERROR, status);
88          status1 = status.transition(TransitionType.NEUTRAL);
89          assertTrue(status == status1);
90  
91          status = DocumentStatus.QUEUED_FOR_PROCESSING;
92          status = status.transition(TransitionType.SUCCESS);
93          assertEquals(DocumentStatus.QUEUED_FOR_PROCESSING_SUCCESS, status);
94  
95          status = status.transition(TransitionType.NEUTRAL);
96          assertEquals(DocumentStatus.DEQUEUED_FOR_PROCESSING, status);
97  
98          status = status.transition(TransitionType.NEUTRAL);

100

103                 DocumentStatus.PROCESSING_KEYWORDS.transition(TransitionType.NEUTRAL));
