8   /**
9    * Unit tests for {@link TermsVectorComparator} covering all {@code getParamValue} branches.
10   */
11  public class TermsVectorComparatorTest {
12  
13      /**
14       * Both terms have a Long tf value – exercises the {@code tf instanceof Long} branch.
15       */
16      @Test
17      public void testCompareWithLongTf() {
18          NamedList<Object> list = new NamedList<>();
19  
20          NamedList<Object> termA = new NamedList<>();
21          termA.add("tf", 5L);
22          list.add("termA", termA);
23  
24          NamedList<Object> termB = new NamedList<>();
25          termB.add("tf", 2L);
26          list.add("termB", termB);
27  
28          OrderOptions opts = new OrderOptions(Order.DESC, "tv.tf");
29          TermsVectorComparator cmp = new TermsVectorComparator(opts, list);
30  
31          // DESC: 5 > 2 → compare("termA","termB") == -1
32          assertEquals(-1, cmp.compare("termA", "termB"));
33      }
34  
35      /**
36       * Both terms have an Integer tf value – exercises the {@code tf instanceof Integer} branch.
37       */
38      @Test
39      public void testCompareWithIntegerTf() {
40          NamedList<Object> list = new NamedList<>();
41  
42          NamedList<Object> termA = new NamedList<>();
43          termA.add("tf", 3);   // Integer
44          list.add("termA", termA);
45  
46          NamedList<Object> termB = new NamedList<>();
47          termB.add("tf", 7);   // Integer
48          list.add("termB", termB);
49  
50          OrderOptions opts = new OrderOptions(Order.ASC, "tv.tf");
51          TermsVectorComparator cmp = new TermsVectorComparator(opts, list);
52  
53          // ASC: 3 < 7 → compare("termA","termB") == -1
54          assertEquals(-1, cmp.compare("termA", "termB"));
55      }
56  
57      /**
58       * A term whose tf field is missing – exercises the {@code tf == null → return 0} branch.
59       * Both values are 0 so result is 1 (equal case returns 1).
60       */
61      @Test
62      public void testCompareWithNullTf() {
63          NamedList<Object> list = new NamedList<>();
64  
65          NamedList<Object> termA = new NamedList<>();
66          // no "tf" entry
67          list.add("termA", termA);
68  
69          NamedList<Object> termB = new NamedList<>();
70          // no "tf" entry
71          list.add("termB", termB);
72  
73          OrderOptions opts = new OrderOptions(Order.ASC, "tv.tf");
74          TermsVectorComparator cmp = new TermsVectorComparator(opts, list);
75  
76          // Both values = 0, equal → returns 1
77          assertEquals(1, cmp.compare("termA", "termB"));
78      }
79  
80      /**
81       * A term that maps to a plain String (not a NamedList) – exercises the
82       * {@code !(value instanceof NamedList) → return 0} branch.
83       */
84      @Test
85      public void testCompareWhenValueNotNamedList() {
86          NamedList<Object> list = new NamedList<>();
87          list.add("termA", "plain string value");   // not a NamedList
88          list.add("termB", "another string");
89  
90          OrderOptions opts = new OrderOptions(Order.ASC, "tv.tf");
91          TermsVectorComparator cmp = new TermsVectorComparator(opts, list);
92  
93          // Both return 0 from getParamValue → equal → returns 1
94          assertEquals(1, cmp.compare("termA", "termB"));
95      }
