17   * Unit tests for {@link DshRestApplication}, its DTOs and {@link SwaggerConfig},

23      /**
24       * Verifies main() hands its arguments to SpringApplication.run().
25       */
26      @Test
27      public void main_delegatesToSpringApplicationRun() {
28          String[] args = {"arg1"};
29          try (MockedStatic<SpringApplication> spring = Mockito.mockStatic(SpringApplication.class)) {
30              DshRestApplication.main(args);
31              spring.verify(() -> SpringApplication.run(DshRestApplication.class, args));
32          }
33      }
34  
35      /**
36       * Verifies DshRestApplication.configure() returns a non-null builder.
37       */
38      @Test
39      public void testConfigure() {
40          SpringApplicationBuilder builder = new SpringApplicationBuilder();
41          SpringApplicationBuilder result = new DshRestApplication().configure(builder);
42          assertNotNull("configure() must return a non-null builder", result);
43      }

45      /**
46       * Verifies DocumentStatusDto accessors.
47       */
48      @Test
49      public void testDocumentStatusDto() {
50          DocumentStatusDto dto = new DocumentStatusDto();
51          dto.setStatus("STATUS");
52          dto.setMessage("MSG");
53          assertEquals("STATUS", dto.getStatus());
54          assertEquals("MSG", dto.getMessage());
55  
56          DocumentStatusDto dto2 = new DocumentStatusDto("S2", "M2");
57          assertEquals("S2", dto2.getStatus());
58          assertEquals("M2", dto2.getMessage());
59      }
