105         when(dao.storeDocument(any(Document.class))).thenAnswer(inv -> {
106             Document d = inv.getArgument(0);
107             return d.getId() != null ? d.getId() : "mock-stored-id";
108         });
109 
110         // Default stub: simulate RabbitMQ ACK – handler receives "sent ok" message
111         // document.getId() is null at enqueueing time, use nullable() matcher
112         doAnswer(inv -> {
113             DocumentEnqueueMessageHandler handler = inv.getArgument(1);
114             handler.handleMessage(
115                     new org.springframework.messaging.support.GenericMessage<>("id sent ok"));
116             return null;
117         }).when(documentQueueService).enqueueDocumentId(
118                 org.mockito.ArgumentMatchers.nullable(String.class), any());

125     /**
126      * A valid document submission must return a non-null token and message.
127      * The RabbitMQ ACK mock drives the document to QUEUED_FOR_INDEXING_SUCCESS.
128      */
129     @Test
130     public void testSubmitDocument() throws Exception {
131         MockMultipartFile firstFile = new MockMultipartFile(
132                 "contents", "bbc-news-1.pdf", "application/pdf",
133                 new FileInputStream(testFile1));
134 
135         startRequest();
136 
137         MvcResult result = mockMvc.perform(
138                 MockMvcRequestBuilders.multipart("/v1/dsh/document/submit")
139                         .file(firstFile)
140                         .param("title", "Russia-Trump: FBI chief Wray defends agency"))
141                 .andExpect(status().isOk())
142                 .andExpect(content().contentType(contentType))
143                 .andExpect(jsonPath("token", notNullValue()))
144                 .andExpect(jsonPath("message", notNullValue()))
145                 .andReturn();
146 
147         // Allow async storeDocumentAndQueueForProcessing to complete
148         synchronized (this) { this.wait(2000); }
149 
150         endRequest();
151 
152         String body = result.getResponse().getContentAsString();
153         logger.info("Submit response: {}", body);
154         TokenDto tDto = new ObjectMapper().readerFor(TokenDto.class).readValue(body);
155 
156         // Stub findDocumentByToken so status endpoint also works in follow-up call
157         Document stored = new Document(
158                 new FileInputStream(testFile1), "Russia-Trump: FBI chief Wray defends agency");
159         stored.transitionStatus(com.mriss.dsh.data.models.TransitionType.SUCCESS);
160         when(dao.findDocumentByToken(tDto.getToken())).thenReturn(stored);
161     }

194     /**
195      * Status endpoint with a known token must return the document status.
196      */
197     @Test
198     public void testStatus() throws Exception {
199         // Submit a document first to obtain a token
200         MockMultipartFile firstFile = new MockMultipartFile(
201                 "contents", "edition.cnn.com-1.pdf", "application/pdf",
202                 new FileInputStream(testFile2));
203 
204         startRequest();
205         MvcResult submitResult = mockMvc.perform(
206                 MockMvcRequestBuilders.multipart("/v1/dsh/document/submit")
207                         .file(firstFile)
208                         .param("title", "Emails show Trump Tower meeting follow-up"))
209                 .andExpect(status().isOk())
210                 .andReturn();
211         synchronized (this) { this.wait(2000); }
212         endRequest();
213 
214         String body = submitResult.getResponse().getContentAsString();
215         TokenDto tDto = new ObjectMapper().readerFor(TokenDto.class).readValue(body);
216 
217         // Prepare a document in success state for the status check
218         Document d = new Document(new FileInputStream(testFile2),
219                 "Emails show Trump Tower meeting follow-up");
220         d.addKeyword(new Keyword("test", 0.5));
221         d.addSentence(new Sentence("test sentence", 0.6, 1));
222         d.transitionStatus(com.mriss.dsh.data.models.TransitionType.NEUTRAL);
223         d.transitionStatus(com.mriss.dsh.data.models.TransitionType.SUCCESS);
224         when(dao.findDocumentByToken(tDto.getToken())).thenReturn(d);
225 
226         startRequest();
227         mockMvc.perform(get("/v1/dsh/document/status/" + tDto.getToken()))
228                 .andExpect(status().isOk())
229                 .andExpect(content().contentType(contentType))
230                 .andExpect(jsonPath("status",
231                         is(DocumentStatus.QUEUED_FOR_INDEXING_SUCCESS.getStatusDescription())))
232                 .andExpect(jsonPath("message",
233                         is(DocumentStatus.QUEUED_FOR_INDEXING_SUCCESS.getStatusMessage())))
234                 .andReturn();
235         endRequest();
236     }
