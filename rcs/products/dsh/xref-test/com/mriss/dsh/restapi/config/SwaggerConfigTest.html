18  /**
19   * Unit tests for {@link SwaggerConfig}'s {@link WebMvcRequestHandlerProvider}
20   * override, with no Spring context.
21   */
22  public class SwaggerConfigTest {
23  
24      /**
25       * Only the mapping backed by AntPathMatcher reaches Springfox; the one using
26       * PathPatternParser is filtered out, so its handler methods are never read.
27       */
28      @Test
29      public void webMvcRequestHandlerProvider_dropsPathPatternParserMappings() {
30          RequestMappingInfoHandlerMapping antPathMapping = mock(RequestMappingInfoHandlerMapping.class);
31          RequestMappingInfoHandlerMapping pathPatternMapping = mock(RequestMappingInfoHandlerMapping.class);
32          when(antPathMapping.getPatternParser()).thenReturn(null);
33          when(pathPatternMapping.getPatternParser()).thenReturn(new PathPatternParser());
34  
35          WebMvcRequestHandlerProvider provider = new SwaggerConfig().webMvcRequestHandlerProvider(
36                  Optional.empty(), mock(HandlerMethodResolver.class),
37                  Arrays.asList(antPathMapping, pathPatternMapping));
38          provider.requestHandlers();
39  
40          verify(antPathMapping).getHandlerMethods();
41          verify(pathPatternMapping, never()).getHandlerMethods();
42      }
43  }
