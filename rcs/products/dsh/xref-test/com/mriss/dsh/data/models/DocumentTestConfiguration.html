10  import org.springframework.context.annotation.Bean;

13  /**
14   * Spring {@link Configuration} providing {@link Document} beans for {@link DocumentTest}.
15   * Loads only the minimal bean definitions needed – no MongoDB or full Boot autoconfiguration required.
16   */
17  @Configuration
18  public class DocumentTestConfiguration {
19  
20      @Bean(name = "docTitleConstructor")
21      public Document getDocumentWithTitle() throws NoSuchAlgorithmException, FileNotFoundException, IOException {
22          Document d = new Document("Russia-Trump: FBI chief Wray defends agency");
23          d.setOriginalFileContents(IOUtils.toByteArray(new FileInputStream(new File("target/test-classes/pdf/bbc-news-1.pdf"))));
24          return d;
25      }
26  
27      @Bean(name = "docTitleContentsConstructor")
28      public Document getDocumentWithTitleAndContents() throws Exception {
29          return new Document(IOUtils.toByteArray(new FileInputStream(new File("target/test-classes/pdf/bbc-news-1.pdf"))), "Russia-Trump: FBI chief Wray defends agency");
30      }
31  
32      @Bean(name = "docTitleContentsFromStreamConstructor")
33      public Document getDocumentWithTitleAndContentsFromStream() throws Exception {
34          return new Document(new FileInputStream(new File("target/test-classes/pdf/bbc-news-1.pdf")), "Russia-Trump: FBI chief Wray defends agency");
35      }
36  
37      @Bean(name = "anotherDocument")
38      public Document getAnotherDocument() throws Exception {
39          return new Document(new FileInputStream(new File("target/test-classes/pdf/edition.cnn.com-1.pdf")), "Emails show Trump Tower meeting follow-up");
40      }
41  }
