47      private MongoDocumentDao dao;

51      @Before
52      public void before() throws Exception {
53          docTitleConstructor = new Document(
54                  IOUtils.toByteArray(new FileInputStream(new File("target/test-classes/pdf/bbc-news-1.pdf"))),
55                  "Russia-Trump: FBI chief Wray defends agency");
56          docTitleConstructor.setId("test-id");
57  
58          // Stub save: return the same document (id already set above)
59          when(repository.save(any(Document.class))).thenReturn(docTitleConstructor);
60  
61          // Stub findById
62          when(repository.findById("test-id")).thenReturn(Optional.of(docTitleConstructor));
63          when(repository.findById("398j")).thenReturn(Optional.empty());
64  
65          // Stub findByFileHash
66          when(repository.findByFileHash(docTitleConstructor.getFileHash()))
67                  .thenReturn(Arrays.asList(docTitleConstructor));
68          when(repository.findByFileHash("1234")).thenReturn(Collections.emptyList());
69  
70          // Stub findByToken
71          when(repository.findByToken(docTitleConstructor.getToken())).thenReturn(docTitleConstructor);
72          when(repository.findByToken("1234")).thenReturn(null);
73      }

75      @Test
76      public void testStoreDocument() {
77          for (int i = 1; i <= 5; i++) {
78              docTitleConstructor.addKeyword(new Keyword("test" + i, ((double) i) + 0.1));
79          }
80          for (int i = 1; i <= 5; i++) {
81              docTitleConstructor.addSentence(new Sentence("test" + i, ((double) i) + 0.21, i + 1));
82          }
83          String id = dao.storeDocument(docTitleConstructor);
84          assertNotNull(id);
85      }
86  
87      @Test
88      public void testFindDocumentById() {
89          String id = dao.storeDocument(docTitleConstructor);
90          assertNotNull(id);
91  
92          Document d = dao.findDocumentById(id);
93          assertNotNull(d);
94          assertTrue(d.equals(docTitleConstructor));
95          assertEquals(DocumentStatus.CREATED, d.getDocumentStatus());
96          logger.info("document(): " + d);
97  
98          d = dao.findDocumentById("398j");
99          assertNull(d);
100     }

102     @Test
103     public void testFindDocumentsByHash() {
104         String id = dao.storeDocument(docTitleConstructor);
105         assertNotNull(id);
106 
107         List<Document> docs = dao.findDocumentsByHash(docTitleConstructor.getFileHash());
108         assertNotNull(docs);
109         assertFalse(docs.isEmpty());
110         for (Iterator<Document> iterator = docs.iterator(); iterator.hasNext();) {
111             Document document = iterator.next();
112             assertTrue(document.equals(docTitleConstructor));
113             assertEquals(DocumentStatus.CREATED, document.getDocumentStatus());
114         }
115 
116         docs = dao.findDocumentsByHash("1234");
117         assertNotNull(docs);
118         assertTrue(docs.isEmpty());
119     }
